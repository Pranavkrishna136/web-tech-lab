// Program 1: Digital Clock using Date Object and setInterval()
function updateClock() {
    var now = new Date();
    var h = now.getHours();
    var m = now.getMinutes();
    var s = now.getSeconds();

    // Pad single digits with leading zero
    h = h < 10 ? "0" + h : h;
    m = m < 10 ? "0" + m : m;
    s = s < 10 ? "0" + s : s;

    document.getElementById("clock-hours").innerText = h;
    document.getElementById("clock-minutes").innerText = m;
    document.getElementById("clock-seconds").innerText = s;
}

// Call updateClock every 1000ms
setInterval(updateClock, 1000);
updateClock();

// Program 2: Dynamic Background Colour every 30 seconds
function changeBackgroundColor() {
    // Generate random values between 0 and 255
    var r = Math.floor(Math.random() * 256);
    var g = Math.floor(Math.random() * 256);
    var b = Math.floor(Math.random() * 256);

    // Apply color to body background style
    var rgbColor = "rgb(" + r + ", " + g + ", " + b + ")";
    document.body.style.backgroundColor = rgbColor;
}

// Call changeBackgroundColor every 30 seconds (30,000 milliseconds)
setInterval(changeBackgroundColor, 30000);

Car Movement Animation

setTimeout() • clearTimeout() • Boundary Checking

Smooth frame animation moving a car across a horizontal track. Uses init() to set the initial position, moveRight() with recursive setTimeout() to move towards the right, and clearTimeout() to stop the car. Includes a boundary collision check so the car stays safely inside the track.

// Program 4: Car Movement Animation using setTimeout & clearTimeout
var car = null;
var timer = null;
var pos = 0;
var speed = 5;

function init() {
    car = document.getElementById("car-image");
    car.style.position = "relative";
    car.style.left = "0px";
    pos = 0;
    clearTimeout(timer);
}

function moveRight() {
    var track = document.getElementById("road-track");
    var maxRight = track.clientWidth - car.offsetWidth - 10;

    if (pos < maxRight) {
        pos += speed;
        car.style.left = pos + "px";
        timer = setTimeout(moveRight, 20); // Repeat every 20ms
    } else {
        clearTimeout(timer); // Boundary reached: stop
    }
}

function stopCar() {
    clearTimeout(timer);
}

window.onload = init;

    { name: "Pranav Krishna", regNo: "URK24CS1273", yob: 2004 },

    { name: "Kevin Joseph",   regNo: "URK24CS1330", yob: 2003 }
];

// Iterate using for loop and classify using if/else
for (var i = 0; i < students.length; i++) {
    var student = students[i];
    var category = "";

    if (student.yob < 2000) {
        category = "Born before 2000";
    } else if (student.yob === 2000) {
        category = "Born in 2000";
    } else {
        category = "Born after 2000";
    }

    console.log(student.name + " (" + student.regNo + "): " + category);
}
